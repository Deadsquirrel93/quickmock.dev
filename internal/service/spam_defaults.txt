# Quickmock default anti-spam patterns (M4.4).
# One RE2 regex per line; lines starting with # and blank lines are ignored.
# Replace this list at runtime with QUICKMOCK_SPAM_PATTERNS_FILE=<path>,
# or disable the filter with QUICKMOCK_SPAM_PATTERNS_FILE=off.
#
# Curation rule: target PROMOTIONAL PHRASING, never bare keywords — users
# legitimately mock arbitrary API payloads ("Casino Royale" is a movie).
# False positives are escapable via QUICKMOCK_SPAM_ALLOW_IPS.

# Pharma spam: drug name only together with buy/order context.
(?i)\b(buy|order|cheap|discount|purchase)\b[^\n]{0,40}\b(viagra|cialis|levitra|tramadol|xanax)\b
(?i)\b(viagra|cialis|levitra|tramadol|xanax)\b[^\n]{0,40}\b(online|without a? ?prescription|for sale)\b

# Casino / gambling promo.
(?i)\b(online casino|best casino|casino bonus|free spins|no deposit bonus|sports betting bonus)\b

# Adult promo with a link (a bare word is not enough).
(?i)\b(porn|xxx|adult dating|escort service)\b[^\n]{0,60}https?://

# SEO / backlink spam.
(?i)\b(buy|sell|cheap|quality)\b[^\n]{0,30}\bbacklinks?\b
(?i)\bincrease your (google |search )?rankings?\b

# Crypto / get-rich scams.
(?i)\b(double|triple) your (bitcoin|btc|crypto|investment|money)\b
(?i)\bguaranteed (profit|returns?|income)\b
(?i)\bmake \$?\d[\d,]{2,} (a|per) (day|week|month) (from home|online)\b

# Security-abuse payloads. These are infrastructure primitives rather than
# useful mock response content and have been abused in public instances.
(?i)\bgopher://
(?i)\b(metadata\.google\.internal|169\.254\.169\.254)\b
(?i)\bFastCgiModule\b[^\n]{0,160}\bscriptProcessor\b
(?i)\bserviceWorker\.register\b[^\n]{0,500}\b(metadata\.google\.internal|169\.254\.169\.254|gopher://)\b

# Backlink-page fingerprints rather than specific advertised domains.
(?i)<h1>[^<]{1,100}</h1>\s*<p>\s*Visit\s+<a\s+href=["']https?://
(?i)\bWebsite:\s*<a\s+href=["']https?://
(?i)\bPublished on \d{4}-\d{2}-\d{2} via Quickmock\b

# Flag-file reads and secret-bearing environment lookups: a mocked flag-file
# path in quotes, or an os.environ subscript/get() naming FLAG specifically —
# never the bare word "flag" (mocks legitimately return {"flag": true}).
(?i)["'](?:/flag(?:\.txt)?|/app/flag|/run/secrets/flag)["']
(?i)\bos\.environ\[["']FLAG["']\]
(?i)\bos\.environ\.get\(["']FLAG["']

# Loader / reverse-shell one-liners.
(?i)\bcurl\b[^\n]{0,80}\|\s*(sh|bash)\b
/dev/tcp/
(?i)\bIEX\s*\(\s*New-Object\s+Net\.WebClient\)
(?i)\bpowershell\b[^\n]{0,60}-enc(odedcommand)?\b\s+[A-Za-z0-9+/=]{8,}

# Cloud metadata endpoints, path form (hostname/IP forms are above).
(?i)/latest/meta-data/
(?i)/computeMetadata/v1/

# Cross-window abuse: deprecated same-origin bypass, and an auto-click loop
# built from an opened window plus a timer (either order).
(?i)\bdocument\.domain\s*=
(?i)\bwindow\.open\b[^\n]{0,300}\bsetInterval\b
(?i)\bsetInterval\b[^\n]{0,300}\bwindow\.open\b

# Fake LLM endpoints that answer with a shell-execution tool call (OpenAI
# tool_calls / function_call, Anthropic tool_use; plain or escaped JSON):
# pointing an agent's base URL at such a mock runs the creator's command.
# Ordinary tool calls (get_weather, search_docs) are untouched.
(?is)"(?:tool_calls|tool_use|function_call)\\?".{0,400}?\\?"name\\?"\s*:\s*\\?"(?:exec|execute|execute_command|shell|shell_exec|local_shell|run_shell|run_shell_command|run_command|bash|sh|zsh|cmd|powershell|terminal)\\?"

# Sandbox network recon (ARP table, local addresses, open sockets, routes).
/proc/net/(?:arp|fib_trie|tcp6?|udp6?|route)\b

# OAuth Client ID Metadata Documents: client_id is the document's own URL, so
# serving one turns the mock URL into an OAuth client identity (found on prod
# impersonating ChatGPT). Registration/token responses use opaque client ids.
(?i)\\?"client_id\\?"\s*:\s*\\?"https?://
